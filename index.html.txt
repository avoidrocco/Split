<script>
(() => {
  const data = {
    mon: {
      label: "Mon", title: "Monday — Upper",
      note: "Balanced chest + back, then delts and arms.",
      exercises: [
        ["Machine Chest Press","3 × 6–10","Last set to failure"],
        ["Lat Pulldown","3 × 6–10","Last set to failure"],
        ["Seated Machine / Cable Row","2 × 8–12","Failure"],
        ["Incline Dumbbell Press","2 × 8–12","0–1 RIR"],
        ["Lateral Raise","3 × 12–20","Failure"],
        ["Cable Triceps Pressdown","2 × 10–15","Failure"],
        ["Dumbbell Curl","2 × 10–15","Failure"]
      ]
    },
    tue: {
      label: "Tue", title: "Tuesday — Lower",
      note: "Primary lower-body day.",
      exercises: [
        ["Hack Squat or Leg Press","3 × 6–10","Last set to failure"],
        ["Dumbbell Romanian Deadlift","2 × 8–12","0–1 RIR"],
        ["Leg Curl","3 × 8–12","Failure"],
        ["Leg Extension","2 × 10–15","Failure"],
        ["Hip Abductor","2 × 12–20","Failure"],
        ["Calf Raise","3 × 8–15","Failure"],
        ["Cable Crunch","3 × 10–15","Failure"]
      ]
    },
    wed: {label:"Wed", title:"Wednesday — Rest", note:"Recovery day.", exercises:[]},
    thu: {
      label: "Thu", title: "Thursday — Push",
      note: "Chest, shoulders and triceps.",
      exercises: [
        ["Incline Machine Chest Press","3 × 8–12","Last set to failure"],
        ["Dumbbell Shoulder Press","2 × 6–10","0–1 RIR"],
        ["Pec Deck","3 × 10–15","Failure"],
        ["Cable Lateral Raise","3 × 12–20","Failure"],
        ["Rope Triceps Pressdown","2 × 10–15","Failure"],
        ["Overhead Cable Triceps Extension","2 × 10–15","Failure"]
      ]
    },
    fri: {
      label: "Fri", title: "Friday — Pull",
      note: "Back, rear delts and biceps.",
      exercises: [
        ["Lat Pulldown / Assisted Pull-Up","3 × 6–10","Last set to failure"],
        ["Chest-Supported Machine Row","3 × 8–12","Last set to failure"],
        ["Single-Arm Cable Row","2 × 8–12","Failure"],
        ["Reverse Pec Deck","3 × 12–20","Failure"],
        ["Straight-Arm Pulldown","2 × 10–15","Failure"],
        ["Preacher Curl Machine","2 × 8–12","Failure"],
        ["Dumbbell Hammer Curl","2 × 10–15","Failure"]
      ]
    },
    satLegs: {
      label:"Sat", title:"Saturday — Legs",
      note:"Use when your legs feel recovered.",
      exercises:[
        ["Leg Press or Hack Squat","3 × 8–12","Last set to failure"],
        ["Dumbbell Bulgarian Split Squat","2 × 8–12 / leg","0–1 RIR"],
        ["Leg Curl","2 × 10–15","Failure"],
        ["Leg Extension","2 × 10–15","Failure"],
        ["Hip Thrust / Glute Machine","2 × 8–12","Last set to failure"],
        ["Calf Raise","3 × 10–20","Failure"],
        ["Ab Machine / Knee Raise","2–3 × 10–20","Failure"]
      ]
    },
    satShoulders: {
      label:"Sat", title:"Saturday — Shoulders + Core",
      note:"Swap this in when your legs need more recovery.",
      exercises:[
        ["Machine Shoulder Press","2 × 8–12","Last set to failure"],
        ["Cable Lateral Raise","3 × 12–20","Failure"],
        ["Reverse Pec Deck","3 × 12–20","Failure"],
        ["Machine Lateral Raise","2 × 10–15","Failure"],
        ["Face Pull","2 × 12–20","Failure"],
        ["Shrug","2 × 10–15","Failure"],
        ["Cable Crunch","3 × 10–15","Failure"],
        ["Hanging Knee / Leg Raise","3 × 8–15","Failure"],
        ["Pallof Press","2 × 10–15 / side","0–1 RIR"]
      ]
    },
    sun: {label:"Sun", title:"Sunday — Rest", note:"Recovery day.", exercises:[]}
  };

  const storageKey = "workout-checklist-v1";
  const defaultState = {day:"mon", saturday:"legs", checks:{}};
  let state = {...defaultState};

  try {
    const saved = JSON.parse(localStorage.getItem(storageKey) || "null");
    if (saved && typeof saved === "object") {
      state = {
        day: saved.day || "mon",
        saturday: saved.saturday === "shoulders" ? "shoulders" : "legs",
        checks: saved.checks || {}
      };
    }
  } catch {}

  const nav = document.getElementById("dayNav");
  const list = document.getElementById("exerciseList");
  const title = document.getElementById("dayTitle");
  const note = document.getElementById("dayNote");
  const count = document.getElementById("count");
  const progressBar = document.getElementById("progressBar");
  const saturdayToggle = document.getElementById("saturdayToggle");

  const order = ["mon","tue","wed","thu","fri","sat","sun"];

  function keyForDay(day) {
    if (day === "sat") return state.saturday === "shoulders" ? "satShoulders" : "satLegs";
    return day;
  }

  function save() {
    try { localStorage.setItem(storageKey, JSON.stringify(state)); } catch {}
  }

  function renderNav() {
    nav.innerHTML = "";
    order.forEach(day => {
      const key = day === "sat" ? keyForDay(day) : day;
      const btn = document.createElement("button");
      btn.type = "button";
      btn.className = "day-btn" + (state.day === day ? " active" : "");
      btn.textContent = data[key].label;
      btn.setAttribute("aria-pressed", state.day === day ? "true" : "false");
      btn.addEventListener("click", () => {
        state.day = day;
        save();
        render();
      });
      nav.appendChild(btn);
    });
  }

  function renderSaturdayToggle() {
    saturdayToggle.hidden = state.day !== "sat";
    saturdayToggle.querySelectorAll("[data-sat]").forEach(btn => {
      const active = btn.dataset.sat === state.saturday;
      btn.classList.toggle("active", active);
      btn.setAttribute("aria-pressed", active ? "true" : "false");
    });
  }

  function renderWorkout() {
    const workoutKey = keyForDay(state.day);
    const workout = data[workoutKey];
    title.textContent = workout.title;
    note.textContent = workout.note;
    list.innerHTML = "";

    if (!workout.exercises.length) {
      const li = document.createElement("li");
      li.className = "rest";
      li.textContent = "Rest, recover, and come back ready for the next session.";
      list.appendChild(li);
      count.textContent = "";
      progressBar.style.width = "0%";
      return;
    }

    let completed = 0;
    workout.exercises.forEach((ex, index) => {
      const id = workoutKey + "-" + index;
      const checked = !!state.checks[id];
      if (checked) completed++;

      const li = document.createElement("li");
      li.className = "exercise" + (checked ? " done" : "");

      const checkbox = document.createElement("input");
      checkbox.type = "checkbox";
      checkbox.checked = checked;
      checkbox.id = id;
      checkbox.setAttribute("aria-label", "Complete " + ex[0]);
      checkbox.addEventListener("change", () => {
        state.checks[id] = checkbox.checked;
        save();
        renderWorkout();
      });

      const details = document.createElement("div");
      const label = document.createElement("label");
      label.className = "name";
      label.htmlFor = id;
      label.textContent = ex[0];

      const meta = document.createElement("div");
      meta.className = "meta";

      const reps = document.createElement("span");
      reps.className = "pill";
      reps.textContent = ex[1];

      const effort = document.createElement("span");
      effort.className = "pill";
      effort.textContent = ex[2];

      meta.append(reps, effort);
      details.append(label, meta);
      li.append(checkbox, details);
      list.appendChild(li);
    });

    const total = workout.exercises.length;
    count.textContent = completed + " / " + total;
    progressBar.style.width = ((completed / total) * 100) + "%";
  }

  function render() {
    renderNav();
    renderSaturdayToggle();
    renderWorkout();
  }

  saturdayToggle.querySelectorAll("[data-sat]").forEach(btn => {
    btn.addEventListener("click", () => {
      state.saturday = btn.dataset.sat;
      save();
      render();
    });
  });

  document.getElementById("resetDay").addEventListener("click", () => {
    const workoutKey = keyForDay(state.day);
    data[workoutKey].exercises.forEach((_, i) => delete state.checks[workoutKey + "-" + i]);
    save();
    renderWorkout();
  });

  document.getElementById("resetWeek").addEventListener("click", () => {
    state.checks = {};
    save();
    renderWorkout();
  });

  render();
})();
</script>
</body>
